title: Steganography or Data Obfuscation Utility Execution (AN2108)
id: f1c2d3e4-a5b6-4c7d-8e9f-0a1b2c3d4e65
status: experimental
description: "Detects the execution of known command-line steganography tools (e.g., steghide, openstego, outguess) commonly used by threat actors to embed or extract hidden payloads inside images or audio files to bypass network monitoring."
author: Krishna Gupta
date: 2026-07-01
logsource:
    category: process_creation
    product: windows
detection:
    selection_steg_images:
        Image|endswith:
            - '\steghide.exe'
            - '\openstego.exe'
            - '\outguess.exe'
    selection_steg_ofn:
        OriginalFileName:
            - 'steghide.exe'
            - 'openstego.exe'
            - 'outguess.exe'
    selection_steg_cli:
        CommandLine|contains:
            - 'embed'
            - 'extract'
            - '-ef'
            - '-cf'
            - '-sf'
            - '-xf'
    filter_installer:
        ParentImage|endswith:
            - '\msiexec.exe'
            - '\setup.exe'
    condition: (selection_steg_images or selection_steg_ofn) and selection_steg_cli and not filter_installer
fields:
    - Image
    - OriginalFileName
    - CommandLine
    - ParentImage
    - User
falsepositives:
    - Developer or security audit environments where steganography tools are tested or used in training exercises.
level: medium
tags:
    - attack.defense_evasion
    - attack.t1001
    - analytic.an2108
    - detection.det0965

title: Direct Volume Access Command Line Detection (AN2109)
id: c3d4e5f6-7a8b-9c0d-1e2f-3a4b5c6d7e80
status: experimental
description: "Detects process execution command-line parameters attempting direct access to raw physical drives, logical drives, or volume shadow copies (e.g., using physical drive paths like \\\\.\\PhysicalDrive or raw logical paths). This is commonly used to bypass OS file lock/permission security controls."
author: Krishna Gupta
date: 2026-07-01
logsource:
    category: process_creation
    product: windows
detection:
    selection_images:
        Image|endswith:
            - '\dd.exe'
            - '\vshadow.exe'
            - '\rawcopy.exe'
            - '\ninja.exe'
            - '\powershell.exe'
            - '\pwsh.exe'
            - '\cmd.exe'
    selection_ofn:
        OriginalFileName:
            - 'dd.exe'
            - 'vshadow.exe'
            - 'rawcopy.exe'
            - 'PowerShell.EXE'
            - 'pwsh.exe'
            - 'Cmd.Exe'
    selection_cli:
        CommandLine|contains:
            - '\\.\PhysicalDrive'
            - '\\.\Volume'
            - '\\.\c:'
            - '\\.\d:'
            - '\\.\e:'
            - '\\.\f:'
    filter_legitimate:
        ParentImage|endswith:
            - '\msiexec.exe'
            - '\setup.exe'
    condition: (selection_images or selection_ofn) and selection_cli and not filter_legitimate
fields:
    - Image
    - OriginalFileName
    - CommandLine
    - ParentImage
    - User
falsepositives:
    - Backup software or disk imaging utilities legitimately reading raw volumes to create system backups.
    - Disk checking tools (chkdsk) run by administrators.
level: high
tags:
    - attack.defense_evasion
    - attack.t1006
    - analytic.an2109
    - detection.det0966

status: experimental
description: "Detects the execution of scripting interpreters (PowerShell, CMD, WScript, CScript) or executable binaries located on removable drive paths (such as D:\\ through Z:\\), which is a common indicator of adversaries running payloads or transferring files from physical media on air-gapped systems."
author: Krishna Gupta
date: 2026-07-01
logsource:
    category: process_creation
    product: windows
detection:
    selection_interpreters:
        Image|endswith:
            - '\powershell.exe'
            - '\pwsh.exe'
            - '\cmd.exe'
            - '\wscript.exe'
            - '\cscript.exe'

    selection_interpreters_ofn:
        OriginalFileName:
            - 'PowerShell.EXE'
            - 'pwsh.exe'
            - 'Cmd.Exe'
            - 'wscript.exe'
            - 'cscript.exe'
            - 'mshta.exe'
    selection_removable_cli:
        CommandLine|re:
            - '(?i)\s+[D-Z]:\\(?!Program Files|Windows)'
    selection_direct_exec:
        Image|re: '(?i)^[D-Z]:\\(?!Program Files|Windows).+\.exe'

    - CommandLine
    - ParentImage

title: Privileged Account or Group Membership Manipulation (AN2114)
id: f1a2b3c4-e5f6-7a8b-9c0d-1e2f3a4b5d95
status: experimental
description: "Detects the addition of users to administrative or privileged groups (Domain Admins, Enterprise Admins, Schema Admins, Local Administrators) or critical modifications to account properties (such as enabling Password Not Required or altering logon permissions) in Windows Security logs."
author: Krishna Gupta
date: 2026-07-01
logsource:
    service: security
    product: windows
detection:
    selection_group_adds:
        EventID:
            - 4728
            - 4732
            - 4756
        # Target privileged group names
        TargetUserName:
            - 'Administrators'
            - 'Domain Admins'
            - 'Enterprise Admins'
            - 'Schema Admins'
            - 'Account Operators'
            - 'Backup Operators'
    selection_account_changes:
        EventID: 4738
        # Look for suspicious flag changes like setting password not required or preauth bypass
        UserAccountControl|contains:
            - 'Password Not Required'
            - "Don't Require Preauth"
            - 'Trusted For Delegation'
            - 'Trusted To Authenticate For Delegation'
    filter_machine_accounts:
        # Exclude standard computer account self-updates
        SubjectUserName|endswith: '$'
    condition: (selection_group_adds or selection_account_changes) and not filter_machine_accounts
fields:
    - EventID
    - TargetUserName
    - SubjectUserName
    - MemberName
    - UserAccountControl
falsepositives:
    - Standard identity management workflows or IT administrators creating, modifying, or promoting accounts during onboarding or maintenance.
level: high
tags:
    - attack.persistence
    - attack.t1098
    - analytic.an2114
    - detection.det0971
