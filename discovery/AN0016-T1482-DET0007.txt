# Atomic Red Team – Prompt Template
Technique: t1482
Analytic ID: an0016
Detection ID: det0007

Description: (taken from rule title - see markdown header)

Commands:
selection_nltest:
        - Image|endswith: '\nltest.exe'
        - OriginalFileName: 'nltest.exe'
    selection_nltest_args:
        CommandLine|contains:
            - '/domain_trusts'
            - '/trusted_domains'
            - '/all_trusts'
    selection_dsquery:
        - Image|endswith: '\dsquery.exe'
        - OriginalFileName: 'dsquery.exe'
    selection_dsquery_args:
        CommandLine|contains: 'trustedDomain'
    selection_powershell:
        - Image|endswith:
            - '\powershell.exe'
            - '\pwsh.exe'
        - OriginalFileName:
            - 'PowerShell.EXE'
            - 'pwsh.dll'
    selection_powershell_args:
        CommandLine|contains:
            - 'Get-ADTrust'
            - 'Get-NetDomainTrust'
            - 'Get-DomainTrust'
            - 'DSEnumerateDomainTrusts'
            - 'NetEnumerateTrustedDomains'
    filter_installer:
        ParentImage|endswith:
            - '\msiexec.exe'
            - '\setup.exe'
    condition: ((selection_nltest and selection_nltest_args) or (selection_dsquery and selection_dsquery_args) or (selection_powershell and selection_powershell_args)) and not filter_installer
fields:
    - Image
    - OriginalFileName
    - CommandLine
    - ParentImage
    - ParentCommandLine
    - CurrentDirectory
    - User
    - LogonId
falsepositives:
    - Domain administrators running trust enumeration during Active Directory health assessments or federation setup tasks.
level: medium
tags:
    - attack.discovery
    - attack.t1482
    - analytic.an0016
    - detection.det0007
