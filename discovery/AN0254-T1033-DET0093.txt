# Atomic Red Team – Prompt Template
Technique: t1033
Analytic ID: an0254
Detection ID: det0093

Description: (taken from rule title - see markdown header)

Commands:
selection_whoami:
        - Image|endswith: '\whoami.exe'
        - OriginalFileName: 'whoami.exe'
    selection_quser:
        - Image|endswith:
            - '\quser.exe'
            - '\query.exe'
        - OriginalFileName:
            - 'quser.exe'
            - 'query.exe'
    selection_net:
        - Image|endswith: '\net.exe'
        - OriginalFileName: 'net.exe'
    selection_net_args:
        CommandLine|contains:
            - ' user'
            - ' localgroup'
    selection_powershell:
        - Image|endswith:
            - '\powershell.exe'
            - '\pwsh.exe'
        - OriginalFileName:
            - 'PowerShell.EXE'
            - 'pwsh.dll'
    selection_powershell_args:
        CommandLine|contains:
            - 'Get-LocalUser'
            - 'Get-ADUser'
            - 'Win32_ComputerSystem'
            - 'Win32_UserAccount'
    filter_installer:
        ParentImage|endswith:
            - '\msiexec.exe'
            - '\setup.exe'
    condition: (selection_whoami or selection_quser or (selection_net and selection_net_args) or (selection_powershell and selection_powershell_args)) and not filter_installer
fields:
    - Image
    - OriginalFileName
    - CommandLine
    - ParentImage
    - ParentCommandLine
    - CurrentDirectory
    - User
    - LogonId
falsepositives:
    - IT administrators executing user discovery commands for asset inventory or helpdesk support activities.
level: low
tags:
    - attack.discovery
    - attack.t1033
    - analytic.an0254
    - detection.det0093

Expected Logs:
- (list of relevant event IDs)

Verification:
- Run the command(s) and confirm log entry appears.
